cmake_minimum_required(VERSION 3.20)
project(lackminer_gpu_control LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "AMD control helper uses the Windows ADLX driver")
endif()
include(FetchContent)
FetchContent_Declare(adlx GIT_REPOSITORY https://github.com/GPUOpen-LibrariesAndSDKs/ADLX.git GIT_TAG 32b5a740d42295c5dfe9026b9f52683da0f3af91)
FetchContent_MakeAvailable(adlx)
add_executable(lackminer-gpu-control amd_control.cpp ${adlx_SOURCE_DIR}/SDK/ADLXHelper/Windows/Cpp/ADLXHelper.cpp ${adlx_SOURCE_DIR}/SDK/Platform/Windows/WinAPIs.cpp)
target_include_directories(lackminer-gpu-control PRIVATE ${adlx_SOURCE_DIR})
target_compile_features(lackminer-gpu-control PRIVATE cxx_std_17)
set_property(TARGET lackminer-gpu-control PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
